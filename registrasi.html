rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // ========================= // HELPER // ========================= function isSignedIn() { return request.auth != null; } function isAdmin() { return isSignedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)) && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin"; } function isOwner() { return isSignedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)) && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "owner"; } function isStaff() { return isAdmin() || isOwner(); } // ========================= // USERS // ========================= match /users/{uid} { // User hanya bisa membaca data dirinya sendiri. // Admin bisa membaca data user. allow read: if isSignedIn() && (request.auth.uid == uid || isAdmin()); // Hanya Admin yang boleh membuat / mengubah / menghapus // data user dan role. allow create, update, delete: if isAdmin(); } // ========================= // PESERTA // ========================= match /peserta/{pesertaId} { // Admin dan Owner dapat melihat data peserta. allow read: if isStaff(); // Peserta TIDAK boleh membuat data langsung // dari browser. // // Registrasi publik akan menggunakan backend // yang sudah kita siapkan. allow create: if false; // ========================= // ADMIN // ========================= // // Admin hanya boleh mengubah data yang berhubungan // dengan pembayaran / approval. allow update: if isAdmin() && request.resource.data.diff(resource.data).affectedKeys().hasOnly([ "statusPembayaran", "nominalPembayaran", "tanggalVerifikasi", "metodePembayaran", "petugasPembayaran", "waktuPembayaran", "petugasVerifikasi", "waktuVerifikasi", "tierPembayaran", "hargaFinal", "catatan" ]); // ========================= // OWNER // ========================= // // Owner hanya boleh melakukan proses check-in. // // Syarat: // 1. Peserta sudah LUNAS // 2. Sebelumnya belum HADIR // 3. Status berubah menjadi HADIR // 4. Benefit menjadi SUDAH DIAMBIL allow update: if isOwner() && resource.data.statusPembayaran == "LUNAS" && resource.data.statusKehadiran != "HADIR" && request.resource.data.statusKehadiran == "HADIR" && request.resource.data.statusBenefit == "SUDAH DIAMBIL" && request.resource.data.diff(resource.data).affectedKeys().hasOnly([ "statusKehadiran", "jamCheckIn", "petugasScan", "waktuScan", "statusBenefit", "catatan" ]); // Tidak boleh dihapus melalui aplikasi. allow delete: if false; } // ========================= // UNIQUE EMAIL // ========================= // // Dipakai untuk mencegah email ganda. // Tidak boleh dibaca / diubah langsung oleh client. match /uniqueEmail/{emailKey} { allow read, write: if false; } // ========================= // UNIQUE WHATSAPP // ========================= // // Dipakai untuk mencegah nomor WhatsApp ganda. // Tidak boleh dibaca / diubah langsung oleh client. match /uniqueWa/{waKey} { allow read, write: if false; } // ========================= // PENGATURAN ACARA // ========================= match /pengaturan/{docId} { // Pengaturan acara boleh dibaca oleh staff. allow read: if isStaff(); // Hanya Admin yang boleh mengubah pengaturan. allow create, update, delete: if isAdmin(); } // ========================= // LOG AKTIVITAS // ========================= match /logAktivitas/{logId} { // Admin dan Owner boleh melihat log. allow read: if isStaff(); // Client tidak boleh membuat / mengubah / menghapus log. allow create, update, delete: if false; } // ========================= // DEFAULT DENY // ========================= // // Semua collection lain ditutup. match /{document=**} { allow read, write: if false; } } }
